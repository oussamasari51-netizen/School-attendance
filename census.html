async function runMovement() { const start = document.getElementById("movStart").value; const end = document.getElementById("movEnd").value; document.getElementById("movLoading").style.display = "block"; const classNameById = {}; allClasses.forEach(c => classNameById[c.id] = c.name); // جلب الوافدين الجدد فقط (شرط صارم: إضافة يدوية صريحة + تاريخ تسجيل محدد) const { data: entries } = await db.from("students") .select("full_name, class_id, enrollment_date") .eq("source", "manual") .not("enrollment_date", "is", null) .gte("enrollment_date", start) .lte("enrollment_date", end) .order("enrollment_date"); const { data: exits } = await db.from("students") .select("full_name, class_id, withdrawal_date, withdrawal_reason") .gte("withdrawal_date", start) .lte("withdrawal_date", end) .order("withdrawal_date"); document.getElementById("movInBody").innerHTML = (entries || []).length ? entries.map(s => `${s.full_name}${classNameById[s.class_id] || ""}${s.enrollment_date}`).join("") : `لا يوجد وافدون جدد خلال هذه الفترة.`; document.getElementById("movOutBody").innerHTML = (exits || []).length ? exits.map(s => `${s.full_name}${classNameById[s.class_id] || ""}${s.withdrawal_reason || ""}${s.withdrawal_date}`).join("") : `لا يوجد مغادرون خلال هذه الفترة.`; document.getElementById("movLoading").style.display = "none"; }
